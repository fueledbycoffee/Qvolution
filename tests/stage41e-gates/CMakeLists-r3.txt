cmake_minimum_required(VERSION 3.16)
project(stage41e_gates_r3 C)
set(L /home/sean/Documents/Codex/Projects/Qvolution-work/parallel/stage41e-gates-fix)
set(S ${L}/superproject)
set(BASE ${L}/../stage41e)
set(B ${BASE}/build)
set(RECIPE ${S}/tests/stage41e-gates/build-r3.py)
add_custom_command(OUTPUT ${L}/build-r3/darling-inherited.c
 COMMAND python3 ${RECIPE} control
 DEPENDS ${S}/src/startup/darling.c ${RECIPE})
foreach(variant detached inherited)
 if(variant STREQUAL detached)
  set(launcher_source ${S}/src/startup/darling.c)
 else()
  set(launcher_source ${L}/build-r3/darling-inherited.c)
 endif()
 add_executable(darling-${variant} ${launcher_source})
 target_compile_definitions(darling-${variant} PRIVATE DARLING MLDR_BUILD _GNU_SOURCE INSTALL_PREFIX="/usr/local")
 target_compile_options(darling-${variant} PRIVATE -std=gnu11 -ggdb -O0)
 target_include_directories(darling-${variant} PRIVATE ${B}/src/include ${S}/src/include ${S}/src/startup ${B}/src/external/darlingserver/include ${BASE}/superproject/src/external/darlingserver/include)
 target_link_libraries(darling-${variant} util)
endforeach()
foreach(framework Automator AVFAudio MediaPlayer)
 if(framework STREQUAL Automator)
  set(changed AMAction AMWorkflow)
 elseif(framework STREQUAL AVFAudio)
  set(changed AVAudioPlayerNode)
 else()
  set(changed MediaPlayer)
 endif()
 set(inputs)
 foreach(stem ${changed})
  list(APPEND inputs ${S}/src/frameworks/${framework}/src/${stem}.m)
 endforeach()
 add_custom_command(OUTPUT ${L}/build-r3/${framework}
  COMMAND python3 ${RECIPE} ${framework}
  DEPENDS ${inputs} ${RECIPE} VERBATIM)
 add_custom_target(${framework}-payload ALL DEPENDS ${L}/build-r3/${framework})
endforeach()
add_custom_command(OUTPUT ${L}/build-r3/stub-forwarding
 COMMAND python3 ${RECIPE} guest
 DEPENDS ${S}/tests/stub-forwarding/guest.m ${S}/tests/stub-forwarding/payload-image.h ${RECIPE})
add_custom_target(stub-forwarding-guest ALL DEPENDS ${L}/build-r3/stub-forwarding)
