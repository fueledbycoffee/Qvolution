cmake_minimum_required(VERSION 3.18)
project(qv_voice_catalog C)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
    set(QV_VOICE_CATALOG_TESTS ${BUILD_TESTING})
else()
    set(QV_VOICE_CATALOG_TESTS ${ENABLE_TESTS})
endif()

find_package(ICU REQUIRED COMPONENTS uc)
find_path(ESPEAK_NG_INCLUDE_DIR espeak-ng/espeak_ng.h REQUIRED)

add_library(qv_voice_catalog_model STATIC catalog.c language.c identity.c)
target_compile_features(qv_voice_catalog_model PRIVATE c_std_11)
target_link_libraries(qv_voice_catalog_model PUBLIC ICU::uc)
target_include_directories(qv_voice_catalog_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(qv_voice_catalog_client STATIC EXCLUDE_FROM_ALL catalog-decode.c catalog-process.c)
target_link_libraries(qv_voice_catalog_client PUBLIC qv_voice_catalog_model)

add_executable(qv-voice-catalog espeak-catalog.c)
target_include_directories(qv-voice-catalog PRIVATE ${ESPEAK_NG_INCLUDE_DIR})
target_link_libraries(qv-voice-catalog PRIVATE qv_voice_catalog_model ${CMAKE_DL_LIBS})
install(TARGETS qv-voice-catalog DESTINATION libexec/darling/usr/libexec/darling COMPONENT gui_stubs)

if(QV_VOICE_CATALOG_TESTS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(frozen-native-test tests/catalog-test.c)
    target_compile_definitions(frozen-native-test PRIVATE _GNU_SOURCE)
    target_link_libraries(frozen-native-test PRIVATE qv_voice_catalog_model)
    add_executable(independent-provider tests/independent-provider.c)
    target_include_directories(independent-provider PRIVATE ${ESPEAK_NG_INCLUDE_DIR})
    target_link_libraries(independent-provider PRIVATE ${CMAKE_DL_LIBS})
    add_library(fault-provider MODULE tests/fault-provider.c)
    target_include_directories(fault-provider PRIVATE ${ESPEAK_NG_INCLUDE_DIR})
    add_executable(fault-catalog espeak-catalog.c)
    target_include_directories(fault-catalog PRIVATE ${ESPEAK_NG_INCLUDE_DIR})
    target_compile_definitions(fault-catalog PRIVATE QV_ESPEAK_LIBRARY="$<TARGET_FILE:fault-provider>")
    target_link_libraries(fault-catalog PRIVATE qv_voice_catalog_model ${CMAKE_DL_LIBS})
    add_dependencies(fault-catalog fault-provider)
    add_executable(unavailable-catalog espeak-catalog.c)
    target_include_directories(unavailable-catalog PRIVATE ${ESPEAK_NG_INCLUDE_DIR})
    target_compile_definitions(unavailable-catalog PRIVATE QV_ESPEAK_LIBRARY="/nonexistent/qv-espeak-absent.so")
    target_link_libraries(unavailable-catalog PRIVATE qv_voice_catalog_model ${CMAKE_DL_LIBS})
    add_executable(allocator-test tests/allocator-test.c)
    target_link_options(allocator-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=free)
    target_link_libraries(allocator-test PRIVATE qv_voice_catalog_model)
    add_test(NAME catalog-allocation-failures COMMAND allocator-test)
    add_executable(script-test tests/script-test.c)
    target_link_libraries(script-test PRIVATE qv_voice_catalog_model)
    add_test(NAME catalog-known-script-policy COMMAND script-test)
    add_executable(client-test tests/client-test.c)
    target_link_libraries(client-test PRIVATE qv_voice_catalog_client)
    add_executable(process-slow tests/process-fixture.c)
    add_executable(process-large tests/process-fixture.c)
    target_compile_definitions(process-large PRIVATE QV_PROCESS_FIXTURE_MODE=1)
    add_executable(process-error tests/process-fixture.c)
    target_compile_definitions(process-error PRIVATE QV_PROCESS_FIXTURE_MODE=2)
    add_test(NAME catalog-client COMMAND client-test $<TARGET_FILE:qv-voice-catalog>
             $<TARGET_FILE:process-slow> $<TARGET_FILE:process-large> $<TARGET_FILE:process-error>)
    set_tests_properties(catalog-client PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    add_executable(decode-allocation-test tests/decode-allocation-test.c)
    target_link_options(decode-allocation-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=free)
    target_link_libraries(decode-allocation-test PRIVATE qv_voice_catalog_client)
    add_test(NAME catalog-decode-allocation-failures COMMAND decode-allocation-test)
    add_test(NAME catalog-provider-comparison COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify-native.py --build $<TARGET_FILE_DIR:qv-voice-catalog>)
    set_tests_properties(catalog-provider-comparison PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
endif()
