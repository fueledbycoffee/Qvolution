cmake_minimum_required(VERSION 3.13)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(darling_surface_native LANGUAGES CXX)
endif()

find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)

add_library(darling_surface_native SHARED descriptor.cpp transport.cpp vulkan.cpp retirement.cpp)
target_compile_features(darling_surface_native PUBLIC cxx_std_17)
target_include_directories(darling_surface_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIRS})
target_link_libraries(darling_surface_native PRIVATE Vulkan::Vulkan Threads::Threads stdc++)

option(DARLING_SURFACE_NATIVE_CONTROLS "Build native controls without GPU execution" OFF)
if(DARLING_SURFACE_NATIVE_CONTROLS)
	enable_testing()
	add_executable(surface-transport-control descriptor.cpp transport.cpp tests/transport-control.cpp)
	add_executable(surface-backend-control descriptor.cpp transport.cpp vulkan.cpp retirement.cpp tests/backend-control.cpp tests/fake-vulkan.cpp)
	add_executable(surface-probe-options-control descriptor.cpp probe/options.cpp probe/options-control.cpp)
	add_executable(surface-probe-fd-control descriptor.cpp transport.cpp probe/options.cpp probe/fd-audit.cpp probe/fd-control.cpp)
	target_link_options(surface-probe-fd-control PRIVATE -Wl,--wrap=readdir)
	foreach(control surface-transport-control surface-backend-control surface-probe-options-control surface-probe-fd-control)
		target_compile_features(${control} PRIVATE cxx_std_17)
		target_include_directories(${control} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIRS})
		target_compile_options(${control} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
		target_link_libraries(${control} PRIVATE Threads::Threads stdc++)
		add_test(NAME ${control} COMMAND ${control})
	endforeach()
endif()

option(DARLING_SURFACE_HARDWARE_PROBE "Build root-only real Vulkan GPU probe (never a CTest)" OFF)
set(DARLING_SURFACE_PROBE_SOURCE_ID "unrecorded" CACHE STRING "Reviewed source revision recorded by the native GPU probe")
if(DARLING_SURFACE_HARDWARE_PROBE)
	add_executable(surface-vulkan-hardware-probe probe/main.cpp probe/options.cpp probe/resources.cpp probe/submission.cpp probe/payload.cpp probe/fd-audit.cpp)
	target_compile_features(surface-vulkan-hardware-probe PRIVATE cxx_std_17)
	target_include_directories(surface-vulkan-hardware-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIRS})
	target_compile_options(surface-vulkan-hardware-probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
	target_compile_definitions(surface-vulkan-hardware-probe PRIVATE SURFACE_PROBE_SOURCE_ID="${DARLING_SURFACE_PROBE_SOURCE_ID}")
	target_link_libraries(surface-vulkan-hardware-probe PRIVATE darling_surface_native Vulkan::Vulkan Threads::Threads stdc++)
endif()
