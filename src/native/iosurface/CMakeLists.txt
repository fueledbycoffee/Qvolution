cmake_minimum_required(VERSION 3.13)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(darling_surface_native LANGUAGES CXX)
endif()

find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)

add_library(darling_surface_native SHARED descriptor.cpp transport.cpp)
target_compile_features(darling_surface_native PUBLIC cxx_std_17)
target_include_directories(darling_surface_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIRS})
target_link_libraries(darling_surface_native PRIVATE Vulkan::Vulkan Threads::Threads stdc++)

option(DARLING_SURFACE_NATIVE_CONTROLS "Build native controls without GPU execution" OFF)
if(DARLING_SURFACE_NATIVE_CONTROLS)
	enable_testing()
	add_executable(surface-transport-control descriptor.cpp transport.cpp tests/transport-control.cpp)
	foreach(control surface-transport-control)
		target_compile_features(${control} PRIVATE cxx_std_17)
		target_include_directories(${control} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIRS})
		target_compile_options(${control} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
		target_link_libraries(${control} PRIVATE Threads::Threads stdc++)
		add_test(NAME ${control} COMMAND ${control})
	endforeach()
endif()
